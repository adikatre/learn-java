public class FrequentFlyerMember {
    private int flyerNumber;
    private String name;
    private int lifeTimeMiles;
    private int statusLevel;

    public FrequentFlyerMember(int flyerNumber, String name, int lifeTimeMiles, int statusLevel) {
        this.flyerNumber = flyerNumber;
        this.name = name;
        this.lifeTimeMiles = lifeTimeMiles;
        this.statusLevel = statusLevel;
    }

    public void addMiles(int miles) {
        lifeTimeMiles += miles;
    }
    
    public void setStatusLevel(int level) {
        statusLevel = level;
    }

    public String getStatusInfo() {
        // frequent flyer or platinum?
        String flyerType;
        if (statusLevel == 1) {flyerType = "frequent flyer"}
        else {flyerType = "platinum"}

        return (Integer.toString(flyerNumber) + " " + name + " is a " + flyerType + " member with " + Integer.toString(lifeTimeMiles) + " miles");
    }
}
