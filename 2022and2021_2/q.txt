public class CombinedTable {

	private SingleTable t1;
	private SingleTable t2;

	private int h1;
	private int h2;

private int seats;

	public CombinedTable(SingleTable t1, SingleTable t2) {
		t1 = this.t1;
		t2 = this.t2;

		h1 = t1.getHeight();
		h2 = t2.getHeight();

		seats = t1.getNumSeats() + t2.getNumSeats() -2;
}

public boolean canSeat(int people) {
	if (seats <= people)
		return true;
	return false;
}

public double getAverageRating() {
	int sum = 0;
	for (int i = 0; i < allReviews.length; i++) {
		sum += allReviews[i].getRating();
}

return (double) sum / allReviews.length;
}

public ArrayList<String> collectComments() {
	ArrayList<String> comments = new ArrayList<String>();

	for (int i = 0; i < allReviews.length; i++) {
		Review r = allReviews[i];
		String f = r.getComment().substring(r.getComment().length()-1, r.getComment().length());
		
		if (r.getComment().contains(“!”) || r.getComment.contains(“.”)) {
			if (f.equals(“!”) || f.equals(“.”))
				comments.add(r.getComment());
			comments.add(r.getComment() + “.”);
}

}

return comments;

}

public void repopulate() {
	for (int r = 0; r < grid.length; r++) {
		for (int c = 0; c < grid[0].length; c++) {
			int value = (int)(Math.random()*10);
			grid[r][c] = value;
}
}
}


public int countIncreasingCols() {
	int iCols = 0;

	for (int c = 0; c < grid[0].length; c++) {
		for (int r = 1; r < grid.length; r++) {
			if (grid[r][c] < grid[r-1][c]) {break;}
			iCols++;
}
}

return iCols;
}
