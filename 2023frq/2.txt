public class Sign {
    private String message;
    private int width;

    public Sign(String message, int width) {
        this.message = message;
        this.width = width;
    }

    public int numberOfLines() {

        int lines = 0;
        int c = 0;

        for (int i = 0; i < this.message.length(); i++) {
            
            lines++;

            if (lines == this.width) {
                c++;
                lines = 0;
            }
        }

        if (lines == 0) {
            return c;
        }
        else {return c+1;}
    }

    public String getLines() {
        String result = "";
        int lines = 0;
        int c = 0;


        if (this.message.equals("")) {return null;}

        for (int i = 0; i < this.message.length(); i++) {
            
            lines++;
            result += this.message.substring(i, i+1);

            if (lines == this.width && (i+1) < this.message.length()) {
                c++;
                lines = 0;
                result += ";"
            }
        }

        return result;
    }
}
