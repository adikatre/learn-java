public int longestHeatWave(double threshold) {
	int longestWave = 0;
	int wave = 0;
	for (int i=0 ; i < temperatures.size() ; i++) {
		for (int j = i; j < temperatures.size() ; j++) {
			if (temperatures.get(j) > threshold) {
				wave++;
				continue;
			}
			break;
		}

		if (wave > longestWave) {
			longestWave = wave;
		}

		wave = 0;

	}
