public static int countVowels(String name) {

    ArrayList<String> vowels = new ArrayList<String>();
    vowels.add("a");
    vowels.add("e");
    vowels.add("i");
    vowels.add("o");
    vowels.add("u");

    int i = 0;
    int v = 0;
    while ((i) != name.length()) {
        if (vowels.contains(name.substring(i, i+1)))
            v++;
        i++;
    }

    return v;
}
